==========|| git ||==========
git checkout -b "rama": crear nueva rama
git push -u origin rama: subir rama nueva al repositorio
git diff: muestra cambios antes del commit
git status: muestra los cambios a subir
git add <archivo1> <archivo2> <carpeta/archivo>: agregar varios archivos
git add .: agregar todos los archivos
git commit -m "mensaje": agregar commit
git push: subir commit
